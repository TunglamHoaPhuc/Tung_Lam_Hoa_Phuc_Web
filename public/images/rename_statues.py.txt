import os
import shutil
import re
import pandas as pd

# ==============================================================================
# 1. ĐƯỜNG DẪN CẤU HÌNH
# ==============================================================================
EXCEL_PATH = r"C:\Users\MY PC\Tung_Lam_Hoa_Phuc_Web\docs\CƠ SỞ DỮ LIỆU TÙNG LÂM HÒA PHÚC_CĂN CHỈNH SƠ BỘ_V3.xlsx"
IMAGE_ROOT_DIR = r"C:\Users\MY PC\Tung_Lam_Hoa_Phuc_Web\public\images\BẢO TƯỢNG PHẬT GIÁO"
OUTPUT_DIR = r"C:\Users\MY PC\Tung_Lam_Hoa_Phuc_Web\public\images"

if not EXCEL_PATH.endswith('.xlsx'):
    EXCEL_PATH += '.xlsx'

os.makedirs(OUTPUT_DIR, exist_ok=True)

# ==============================================================================
# 2. HÀM CHUẨN HÓA CHUỖI TIẾNG VIỆT
# ==============================================================================
def normalize_string(text):
    if not isinstance(text, str):
        return ""
    text = text.lower()
    text = re.sub(r'[àáạảãâầấậẩẫăằắặẳẵ]', 'a', text)
    text = re.sub(r'[èéẹẻẽêềếệểễ]', 'e', text)
    text = re.sub(r'[ìíịỉĩ]', 'i', text)
    text = re.sub(r'[òóọỏõôồốộổỗơờớợởỡ]', 'o', text)
    text = re.sub(r'[ùúụủũưừứựửữ]', 'u', text)
    text = re.sub(r'[ỳýỵỷỹ]', 'y', text)
    text = re.sub(r'[đ]', 'd', text)
    text = re.sub(r'[^a-z0-9]', '', text)
    return text

# ==============================================================================
# 3. TỰ ĐỘNG DÒ HÀNG TIÊU ĐỀ TRONG FILE EXCEL
# ==============================================================================
print("📂 Đang kiểm tra và dò hàng tiêu đề file Excel...")

xls = pd.ExcelFile(EXCEL_PATH)
target_sheet = None
for sheet in xls.sheet_names:
    if 'BẢO TƯỢNG' in sheet.upper() or 'TUONG' in sheet.upper():
        target_sheet = sheet
        break
if not target_sheet:
    target_sheet = xls.sheet_names[0]

# Đọc thử file để tìm hàng chứa chữ 'MÃ' và 'TÊN TƯỢNG'
raw_df = pd.read_excel(EXCEL_PATH, sheet_name=target_sheet, header=None)
header_row_index = 0

for idx, row in raw_df.iterrows():
    row_text = " ".join([str(val) for val in row.values if pd.notna(val)]).upper()
    if 'MÃ' in row_text and ('TÊN TƯỢNG' in row_text or 'TƯỢNG PHÁP' in row_text):
        header_row_index = idx
        break

# Đọc chính xác từ hàng tiêu đề dò được
df = pd.read_excel(EXCEL_PATH, sheet_name=target_sheet, header=header_row_index)
df.columns = [str(c).strip() for c in df.columns]

# Tìm đúng tên cột Mã và tên cột Tiêu đề
ma_col = [c for c in df.columns if 'MÃ' in c.upper()][0]
title_col = [c for c in df.columns if 'TÊN TƯỢNG' in c.upper() or 'TƯỢNG PHÁP' in c.upper()][0]

mapping_dict = {}
for _, row in df.iterrows():
    code = str(row[ma_col]).strip().lower()
    title = str(row[title_col]).strip()
    if code != 'nan' and title != 'nan':
        norm_title = normalize_string(title)
        if norm_title:
            mapping_dict[norm_title] = code

print(f"✅ Đã dò tìm thành công cột '{ma_col}' và '{title_col}'. Nạp {len(mapping_dict)} bảo tượng.")

# ==============================================================================
# 4. QUÉT ẢNH VÀ ĐỔI TÊN ĐỒNG BỘ
# ==============================================================================
success_count = 0
not_found_list = []

print("\n🔍 Đang quét các thư mục con và đồng bộ ảnh...")
for root, dirs, files in os.walk(IMAGE_ROOT_DIR):
    for filename in files:
        if filename.lower().endswith(('.jpg', '.jpeg', '.png', '.webp')):
            old_filepath = os.path.join(root, filename)
            
            name_without_ext = os.path.splitext(filename)[0]
            norm_filename = normalize_string(name_without_ext)
            ext = os.path.splitext(filename).lower()
            
            matched_code = None
            for norm_title, code in mapping_dict.items():
                if norm_title in norm_filename or norm_filename in norm_title:
                    matched_code = code
                    break
            
            if matched_code:
                new_filename = f"{matched_code}{ext}"
                new_filepath = os.path.join(OUTPUT_DIR, new_filename)
                
                shutil.copy2(old_filepath, new_filepath)
                print(f" [OK] {filename}  --->  {new_filename}")
                success_count += 1
            else:
                not_found_list.append(filename)

print("\n" + "="*60)
print(f"🎉 TỔNG KẾT: Đã đồng bộ {success_count} ảnh vào '{OUTPUT_DIR}'.")
if not_found_list:
    print(f"⚠️ Có {len(not_found_list)} ảnh chưa khớp được tên:")
    for f in not_found_list[:10]:
        print(f"  - {f}")
